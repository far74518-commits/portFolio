// 🎯 Clean Architecture: Repository Pattern & State Orchestration
class ProductRepository implements IProductRepository {
  final ApiClient _apiClient;
  final LocalCache _cache;

  ProductRepository({
    required ApiClient apiClient,
    required LocalCache cache,
  })  : _apiClient = apiClient,
        _cache = cache;

  @override
  Future<Result<List<Product>>> fetchFeaturedProducts() async {
    try {
      final cached = await _cache.getProducts();
      if (cached.isNotEmpty) return Result.success(cached);

      final response = await _apiClient.get('/api/v1/products/featured');
      final products = (response.data as List)
          .map((json) => Product.fromJson(json))
          .toList();

      await _cache.saveProducts(products);
      return Result.success(products);
    } catch (e, stackTrace) {
      return Result.failure(AppException.from(e, stackTrace));
    }
  }
}
// ⚡ Clean Architecture: Reactive State Engine with Dynamic Cache
import { useState, useEffect, useCallback } from 'react';

export const useArchitectureEngine = (apiEndpoint, options = {}) => {
  const [state, setState] = useState({
    data: null,
    isLoading: true,
    error: null,
    latency: 0
  });

  const executePipeline = useCallback(async () => {
    const startTime = performance.now();
    try {
      setState(prev => ({ ...prev, isLoading: true }));
      const res = await fetch(apiEndpoint, {
        headers: { 'Content-Type': 'application/json', ...options.headers }
      });
      if (!res.ok) throw new Error(`HTTP Error: ${res.status}`);
      
      const payload = await res.json();
      const latency = Math.round(performance.now() - startTime);

      setState({ data: payload, isLoading: false, error: null, latency });
    } catch (err) {
      setState({ data: null, isLoading: false, error: err.message, latency: 0 });
    }
  }, [apiEndpoint]);

  useEffect(() => { executePipeline(); }, [executePipeline]);
  return { ...state, refetch: executePipeline };
};
// 🔥 Firebase Cloud Architecture: Secure Atomic Transaction
import { getFirestore, runTransaction, doc, serverTimestamp } from 'firebase/firestore';

export async function processOrderTransaction(orderId, customerId, cartItems) {
  const db = getFirestore();
  const orderRef = doc(db, 'orders', orderId);
  const userRef = doc(db, 'users', customerId);

  return await runTransaction(db, async (transaction) => {
    const userDoc = await transaction.get(userRef);
    if (!userDoc.exists()) throw new Error("Customer profile not found!");

    // Validate inventory & calculate total securely
    let subtotal = 0;
    for (const item of cartItems) {
      const itemRef = doc(db, 'inventory', item.id);
      const itemDoc = await transaction.get(itemRef);
      if (itemDoc.data().stock < item.quantity) {
        throw new Error(`Insufficient stock for: ${item.name}`);
      }
      transaction.update(itemRef, { stock: itemDoc.data().stock - item.quantity });
      subtotal += itemDoc.data().price * item.quantity;
    }

    // Commit atomic order record
    transaction.set(orderRef, {
      customerId,
      totalAmount: subtotal,
      status: 'CONFIRMED',
      createdAt: serverTimestamp()
    });

    return { success: true, orderId, subtotal };
  });
}
